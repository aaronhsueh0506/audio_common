common — 共用 DSP 層(FFT、STFT/ISTFT、HPF、重取樣)

概要

common 是 AEC / NR / Audio_ALG pipelines 共用的 DSP 底層,位於 sibling checkout audiolib/common/。消費端加入 -I common/include -I common,連結 libaudio_common.a,TU 帶 -ffp-contract=off -fno-math-errno。空間相關模組(SRP、BF、VAD)在同 repo 的 doa/、bf/、VAD/,見 空間鏈。

模組標頭生命週期呼叫端要保存延遲
FFTfft_wrapper.hheap(create/destroy)或 pool(get_mem_size+init)handle(可被多個模組共借,呼叫須循序)0,無 warm-up
窗audio_window.h無狀態:ac_window_build窗表(自備 N 個 float)—
STFT / ISTFTaudio_stft.h只有 pool,沒有 destroyAcStft、AcIstft恆等串接 = hop = fft_size/2 樣本
HPFhpf.hheap 或 poolHpf(整段串流存活)無 bulk 延遲
重取樣audio_resampler.hheap 或 poolAudioResampler(整段串流存活)有,查 audio_resampler_latency_input_frames
標頭型核心simd_kernels.h 等header-only,無狀態——
  • 一個公開 API、一個 FFT 後端:NE10 是唯一後端,依 ARM32／ARM64 編譯目標自動選路;有 NEON 且 SIMD=1 時走 NEON FFT,其餘走 C FFT;BACKEND=ne10 是預設也是唯一接受的值。SIMD=0 保留後端、改走純量 C 核心,與 NEON 路徑只在浮點捨入內一致。同一 SIMD 設定下 heap 與 static 路徑 byte-equal(selftest 實測)。
  • FP 政策:-ffp-contract=off 是四 repo 統一政策,每個 Makefile 把它放在 CFLAGS 最後,並在 parse 期擋掉 -ffast-math/-Ofast/自帶 -ffp-contract 的覆寫;scripts/audit_fp_contract.sh 用反組譯檢查預期為純量的目的檔沒有自動 FMA。
  • 精度:音訊與頻譜為 float32;HPF 參數邊界驗證以 double 比較。
  • sqrt-Hann 慣例:window = sqrt(periodic Hann),50% overlap 即 COLA、無需再正規化。ac_window_build(w, n) 提供這張表,公式 sqrtf(0.5f * (1.0f - cosf(2.0f * pi * k / N)));AEC c_impl/src/pbfdkf.c、NR c_impl/example/main_mem.c 與 Audio_ALG 的 DFN rate bridge 都呼叫它。

如何呼叫使用

通則(每個有狀態模組相同):(1) 填設定;(2) X_get_mem_size(...),0 永遠代表失敗,不是「不需要記憶體」;(3) 配置 16-byte 對齊的 pool;(4) X_init(pool, bytes, ...),NULL 代表設定不合法、未對齊或 pool 太小;(5) 每個 block 呼叫處理函式;(6) 換串流時 X_reset;(7) X_destroy(pool 路徑是 no-op,可重複呼叫;heap 路徑只能一次)。物件要活過整段串流,不要逐 block 重建。heap 路徑把 (2)–(4) 換成 X_create(...)。以下範例都已用 -Wall -Wextra -Werror 編譯並執行過。

FFT

#include "fft_wrapper.h"

/* 一支 512 點 FFT handle,放在呼叫端的 pool;fft_get_mem_size 已含 NE10 內部 cfg/twiddle */
static unsigned char g_fft_pool[16 * 1024] __attribute__((aligned(16)));

FftHandle *fft_open(void)
{
    size_t bytes = fft_get_mem_size(512);            /* 0 = fft_size 不是 [16, 8192] 內的 2 的冪 */
    if (bytes == 0 || bytes > sizeof g_fft_pool) return NULL;
    return fft_init(g_fft_pool, bytes, 512);          /* NULL = 未對齊 / pool 太小 */
}

void fft_hop(FftHandle *h, const float *time_in /* [512] */, float *time_out /* [512] */)
{
    Complex spec[257];                                /* fft_get_n_freqs(h) = 512/2 + 1 */
    fft_forward(h, time_in, spec);                    /* 未正規化 rfft;輸入輸出不得重疊 */
    /* ...頻域處理... */
    fft_inverse(h, spec, time_out);                   /* 已含 1/N,不要再除一次 */
}
/* 結束時 fft_destroy(h):pool 路徑是 no-op */

HPF → STFT → 頻域處理 → ISTFT(pipeline 的典型 hop)

#include <string.h>
#include "audio_stft.h"
#include "hpf.h"

enum { FFT_N = 512, HOP = FFT_N / 2, N_FREQS = FFT_N / 2 + 1 };

typedef struct { Hpf *hpf; AcStft *an; AcIstft *syn; } Chain;

/* 三塊 16-byte 對齊的 static pool;實際大小由 *_get_mem_size 查詢,放不下就失敗 */
static unsigned char g_hpf_pool[64]      __attribute__((aligned(16)));
static unsigned char g_an_pool[24 * 1024]  __attribute__((aligned(16)));
static unsigned char g_syn_pool[24 * 1024] __attribute__((aligned(16)));

int chain_open(Chain *c, int sample_rate)
{
    AcStftConfig cfg;
    ac_stft_config_default(&cfg, FFT_N);       /* center 0、AC_OLA_ACCUMULATE、scale 1.0f、自帶 FFT 與窗 */
    size_t sa = ac_stft_get_mem_size(&cfg), ss = ac_istft_get_mem_size(&cfg);   /* 0 = 設定不合法 */
    if (sa == 0 || ss == 0 || sa > sizeof g_an_pool || ss > sizeof g_syn_pool ||
        hpf_get_mem_size() > sizeof g_hpf_pool) return -1;
    c->hpf = hpf_init(g_hpf_pool, sizeof g_hpf_pool, 80.0f, sample_rate);     /* NULL = cutoff 不合法 */
    c->an  = ac_stft_init(g_an_pool, sa, &cfg);
    c->syn = ac_istft_init(g_syn_pool, ss, &cfg);
    return (c->hpf && c->an && c->syn) ? 0 : -1;
}

/* 每個 hop:HOP 個樣本進、HOP 個樣本出(整體延遲 = HOP 個樣本) */
int chain_hop(Chain *c, const float *in, float *out)
{
    float x[HOP];
    Complex spec[N_FREQS];
    memcpy(x, in, sizeof x);
    hpf_process(c->hpf, x, HOP);                       /* 就地;mono */
    if (ac_stft_push(c->an, x, spec) != 1) return -1;  /* center=0:恆 1 幀 */
    /* ...在 spec[0..N_FREQS) 上做頻域處理... */
    return ac_istft_push(c->syn, spec, out) == HOP ? 0 : -1;
}

void chain_reset(Chain *c)                         /* 換串流時;不要在一般 hop 邊界呼叫 */
{
    ac_stft_reset(c->an); ac_istft_reset(c->syn); hpf_reset(c->hpf);
}

center = 1 與 AC_OLA_ENV_NORM(torch 語意)的差異:第 1 次 push 回 0 幀、第 2 次回 2 幀(spec 要容納 2 幀,第 f 幀在 spec + f * N_FREQS);ENV_NORM 的第 1 幀 ac_istft_push 回 0;收尾用 ac_stft_flush_planar 補尾端 reflect 幀、ac_istft_flush 吐尚在 overlap 內的 hop 個樣本,之後呼叫 reset。完整表格見 common 細節。

重取樣

#include "audio_resampler.h"

static unsigned char g_rs_pool[2048] __attribute__((aligned(16)));

AudioResampler *rs_open(void)                          /* 48 kHz -> 16 kHz,單聲道 */
{
    size_t bytes = audio_resampler_get_mem_size(48000, 16000, 1);   /* 0 = 取樣率/聲道數不合法 */
    if (bytes == 0 || bytes > sizeof g_rs_pool) return NULL;
    return audio_resampler_init(g_rs_pool, bytes, 48000, 16000, 1);
}

/* 送進 in_frames 個 frame(交錯 float32);回傳寫進 out 的 frame 數,失敗回 -1 */
int rs_block(AudioResampler *rs, const float *in, int in_frames, float *out, int out_cap)
{
    int total = 0;
    if (audio_resampler_output_bound(rs, in_frames) > out_cap) return -1;   /* 容量不夠就不要硬送 */
    while (in_frames > 0) {
        int used = 0, made = 0;
        if (audio_resampler_process(rs, in, in_frames, out + total, out_cap - total,
                                    &used, &made) != 0) return -1;
        if (used == 0) break;                          /* 保險:沒前進就停 */
        in += used; in_frames -= used; total += made;
    }
    return total;
}
常見錯誤:輸出容量不要按比例自己算(48k→16k、480 frame 的 output_bound 是 161、實際產出 160);不同取樣率時 in/out 不可重疊;input_frames 是 frame 數不是樣本數;audio_resampler_process 回 0 才是成功;不要在一般 block 邊界呼叫 reset。

延遲摘要與 warm-up

模組延遲warm-up
FFT0 樣本(單呼叫內完成)無
AcStft→AcIstft 恆等串接hop = fft_size/2 樣本(center 0/1 相同;test_stft 以 impulse 量測)center=1:第 1 次 push 無幀;ENV_NORM:第 1 個合成幀不吐樣本
HPF無 bulk 延遲無(初始 state 0)
