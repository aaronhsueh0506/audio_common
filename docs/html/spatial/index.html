audiolib 文件

空間鏈 — VAD、SRP-PHAT、波束器的串接

audiolib 的 VAD/、doa/、bf/ 三個模組。逐元件的參數與函式表見 VAD、DOA/SRP、BF。

概要

每支麥克風的 FFT 輸出含 F = NFFT/2+1 個複數 bin。vad_api_process(v, X[0]) 讀一路頻譜，doa_step 與 bf_process_with_weights 讀整個麥克風陣列 X。空間模組共用 Complex 型別，連結順序為 libdoa.a libbf.a libvad.a libspatial_common.a -lm。

 麥克風 x M → (每路 AEC / 前處理) → X[0..M-1]  (每路 F 個 bin)
                                     |
          +--------------------------+-------------------------------+
          v                          v                               v
  vad_api_process(X[0])      doa_step(X, mask, raw, out)     bf_process_with_weights(X, doa_s, ..., mask)
   -> raw / out / mask  --->   -> doa_s (弧度,NaN=尚無)  --->    -> 波束後頻譜 beam[F] + 等效權重 w[M*F]
  • 專案內用到 SRP 與 BF 的是 Audio_ALG 的 sample beam helper(Audio_ALG/pipelines/4ch/array_sample_beam.h 與 array_sample_beam.c):它建立 SRP 與 BF、讓 BF 借用 srp->a_array,並在 aec_process_linear() 與 aec_process_residual() 之間每 hop 跑 doa_step() 與 bf_process_with_weights()。VAD 旗標與頻率遮罩由呼叫端帶入 array_sample_beam_process_beam() / array_sample_beam_process_with_activity();array_sample_beam_process() 則改用 helper 內建的 energy VAD(vad_raw = vad_out、不帶遮罩),由 ArraySampleBeamParams 的 auto_vad_threshold_dbfs、auto_vad_snr_ratio、auto_vad_hangover_frames 決定。波束後頻譜與等效權重再經 array_sample_beam_fuse()(內部呼叫 aec_process_residual())交給後級。專案內沒有任何 pipeline 實例化 VADApi(只有 tests/ 與 bench-masker 在用)。本頁的 VAD 步驟是這三個模組的設計用法,不是已量產的接線。呼叫順序見 4ch pipeline。
  • 遮罩極性在兩個標頭都有寫:mask[f] 非 0 表示「SRP 用這個 bin」,同一個遮罩傳給 bf_process_with_weights 時表示「凍結這個 bin 的 RLS 更新」,兩邊極性相反;array_sample_beam.h 的 frequency_mask 也是這樣定義:NULL 或 n_freqs 個 int,1 表示該 bin 供 SRP 使用,同時凍結該 bin 上的 BF 更新。

如何呼叫使用

順序:(1) 各模組 *_get_mem_size,任一回 0 代表設定被拒;(2) 配置 16-byte 對齊的 pool;(3) vad_api_init、srp_init、bf_init(bf_init 要 SRP 的 steering 表,所以在 srp_init 之後);(4) 每個 hop 依序 vad_api_process → doa_step → bf_process_with_weights;(5) 結束時 BF 先於 SRP 銷毀。本例採 4 支麥克風、16 kHz、NFFT 256。VAD 的閾值可依收音環境調整。

#include <math.h>
#include <stdlib.h>
#include <string.h>
#include "srp.h"
#include "bf.h"
#include "vad_api.h"

enum { MICS = 4, FFT_N = 256, BINS = FFT_N / 2 + 1, ANGLES = 72, SAMPLE_RATE = 16000 };

typedef struct {
    SRP *srp; BF *bf; VADApi *vad;
    void *srp_pool, *bf_pool, *vad_pool;
} Spatial;

static void *pool_alloc(size_t bytes)                 /* 16-byte 對齊;嵌入式可換成 static 陣列 */
{
    void *p = NULL;
    return (bytes && posix_memalign(&p, 16, bytes) == 0) ? p : NULL;
}

void spatial_close(Spatial *s)
{
    vad_api_destroy(s->vad);
    bf_destroy(s->bf);                                /* BF 借用 SRP 的 steering,先銷毀 BF */
    srp_destroy(s->srp);
    free(s->vad_pool); free(s->bf_pool); free(s->srp_pool);
    memset(s, 0, sizeof *s);
}

/* mic_x/mic_y:各 mic 在陣列平面上的位置(公尺);DOA 角 θ 的平面波延遲 τ = -(x·cosθ + y·sinθ)/c */
int spatial_open(Spatial *s, float *mic_x, float *mic_y)
{
    ArrayGeometry geom = { MICS, mic_x, mic_y };       /* 只在 srp_init 內被讀取 */
    SRP_Config sc;  BF_Config bc;  VADApiConfig vc;
    memset(s, 0, sizeof *s); memset(&sc, 0, sizeof sc); memset(&bc, 0, sizeof bc); memset(&vc, 0, sizeof vc);

    sc.M = MICS; sc.F = BINS; sc.num_angles = ANGLES;                /* 衍生量:由陣列與 FFT 格點決定 */
    sc.sr = (float)SAMPLE_RATE; sc.NFFT = (float)FFT_N; sc.c = 343.0f;
    sc.low_freq = 300.0f; sc.high_freq = 7000.0f;                    /* 搜尋頻帶 */
    sc.enable_smoothing = 1; sc.switch_consec = 3;
    sc.angle_tol = 10.0f * 3.14159265f / 180.0f; sc.update_interval = 2;

    bc.enable = 1; bc.lambda = 0.995f; bc.mu = 0.1f; bc.adapt_interval = 1;   /* 自動 DOA、不固定波束 */

    vc.backend = VAD_BACKEND_MASKER;                                 /* 以下數值是示意,不是產品預設 */
    vc.masker_cfg.NFFT = FFT_N; vc.masker_cfg.sr = SAMPLE_RATE;
    vc.masker_cfg.E_alpha_up = 0.999f; vc.masker_cfg.E_alpha_down = 0.97f; vc.masker_cfg.margin_dB = 6.0f;
    vc.masker_cfg.low_freq = 200.0f; vc.masker_cfg.high_freq = 8000.0f;
    vc.masker_cfg.M_alpha = 0.995f; vc.masker_cfg.spp_thr = 0.9f; vc.masker_cfg.spp_upd_thr = 0.3f;
    vc.masker_cfg.enable_freq_smooth = 1; vc.masker_cfg.smooth_size = 5;      /* 必須為正奇數 */
    vc.masker_cfg.enable_time_smooth = 1; vc.masker_cfg.T_alpha = 0.6f;
    vc.masker_cfg.enable_energy = vc.masker_cfg.enable_spp = vc.masker_cfg.enable_band = 1;
    vc.vad_cfg.F = BINS;                                             /* 必須等於 NFFT/2+1 */
    vc.vad_cfg.mode = 1; vc.vad_cfg.mask_thr = 0.2f; vc.vad_cfg.min_bins = 3;
    vc.vad_cfg.enable_median = 1; vc.vad_cfg.median_k = 5;           /* 只接受 5/7/9 */
    vc.vad_cfg.enable_smooth = 1; vc.vad_cfg.hangover = 8;

    size_t srp_bytes = srp_get_mem_size(&sc), bf_bytes = bf_get_mem_size(MICS, BINS),
           vad_bytes = vad_api_get_mem_size(&vc);                    /* 任一為 0 = 設定被拒絕 */
    if (!srp_bytes || !bf_bytes || !vad_bytes) return -1;
    s->srp_pool = pool_alloc(srp_bytes); s->bf_pool = pool_alloc(bf_bytes); s->vad_pool = pool_alloc(vad_bytes);
    if (!s->srp_pool || !s->bf_pool || !s->vad_pool) goto fail;

    s->srp = srp_init(s->srp_pool, srp_bytes, &sc, &geom);           /* NULL = cfg/geom 不合法、未對齊或太小 */
    if (!s->srp) goto fail;
    s->bf = bf_init(s->bf_pool, bf_bytes, MICS, BINS, ANGLES, s->srp->a_array, &bc);   /* a_array 是借用 */
    s->vad = vad_api_init(&vc, s->vad_pool, vad_bytes);
    if (!s->bf || !s->vad) goto fail;
    return 0;
fail:
    spatial_close(s);
    return -1;
}

/* 每個 hop 一次。X[m] = 第 m 路的 BINS 個頻譜 bin(例如 AEC lane 的線性輸出頻譜)。
 * beam[BINS] 得到 BF 輸出頻譜;eff_w[MICS*BINS] 得到本 hop 更新前的等效權重(不需要時傳 NULL)。 */
void spatial_hop(Spatial *s, const Complex *const *X, Complex *beam, Complex *eff_w)
{
    vad_api_process(s->vad, X[0]);                                   /* 1. 用一路頻譜做 VAD */
    int vad_raw = vad_api_get_raw(s->vad);                           /*    未經中值與 hangover */
    int vad_out = vad_api_get(s->vad);                               /*    經中值與 hangover */
    const int *mask = vad_api_get_mask(s->vad);                      /*    非 0 = SRP 用該 bin、BF 凍結該 bin */
    doa_step(s->srp, X, mask, vad_raw, vad_out);                     /* 2. DOA(vad_raw 為 1 才跑 SRP) */
    bf_process_with_weights(s->bf, X, doa_get_smooth(s->srp),        /* 3. 波束:DOA 為 NaN 時直通第 0 路 */
                            vad_out ? 0 : 1, mask, beam, eff_w);     /*    只在 vad_out == 0 時允許 RLS 更新 */
}
情況行為
任一 *_get_mem_size 回 0設定被拒;不要 init。SRP 看 M>1、F>1、num_angles>0 與 sr、NFFT、c 有限且 >0、low_freq/high_freq 有限
srp_init/bf_init/vad_api_init 回 NULL設定不合法、pool 未對齊或太小(srp_init 另檢查 geom->M == cfg->M 且 x、y 非 NULL)
doa_step、vad_api_process 收到 NULL靜默返回
bf_process_with_weights、srp()、srp2doa()不檢查 NULL;bf_init 也不檢查 a_array 的實際形狀(必須是 [num_angles][M][F],num_angles 與 SRP 相同),呼叫端負責
非有限的頻譜值BF 會重置受影響 bin 的 RLS 狀態(bf_get_bin_resets 計數);SRP 與 VAD 沒有此防護,請在進 FFT 前攔截
重置srp_reset、bf_reset;VAD 三層(masker、vad、vad_api)都沒有 reset 函式,要重置就對同一塊 pool 再呼叫 vad_api_init(init 會先把該塊清零)
執行緒每個物件有可變狀態,同一物件不可同時被兩條執行緒使用

流程時序圖

 Application            VADApi             SRP              BF
     |                    |                 |                |
 ─── 初始化(一次)───────────────────────────────────────────────────────────────
     |-- vad_api_get_mem_size(&vc) ---->|                 |     0 = F != NFFT/2+1、median_k 非 5/7/9、smooth_size 非正奇數…
     |-- srp_get_mem_size(&sc) ---------------------->|     |     0 = 設定被拒
     |-- bf_get_mem_size(M, F) ------------------------------->|     0 = M <= 0 或 F <= 0(與 num_angles、cfg 無關)
     |   (配置三塊 16-byte 對齊 pool)
     |-- vad_api_init(&vc, mem, n) -->|                 |                |
     |-- srp_init(mem, n, &sc, &geom) ------------->|  建 steering 與 pair 表(geom 只在此被讀)
     |-- bf_init(mem, n, M, F, A, srp->a_array, &bc) -------------->|  borrow a_array
 ─── 穩態 hop(每 hop 一次)──────────────────────────────────────────────────────
     |-- vad_api_process(v, X[0]) ->|  masker_step → vad_step
     |<- vad_api_get_raw / vad_api_get / vad_api_get_mask -|
     |-- doa_step(srp, X, mask, vad_raw, vad_out) ----->|  vad_raw 且 counter % update_interval == 0:
     |                    |                 |             srp() → srp2doa() → doa_smoother_update(…, vad_out)
     |                    |                 |             否則 doa_raw = NaN、doa_s 維持上一個值
     |-- doa_get_smooth(srp) -------------->|  → doa_s
     |-- bf_process_with_weights(bf, X, doa_s, allow_adapt, mask, beam, w) ->|
     |                    |                 |    enable=0 或 doa_s=NaN: beam = X[0],不更新;
     |                    |                 |    否則 DAS − wa 分支;allow_adapt 且 frame_idx % adapt_interval == 0 才更新 RLS
     |<- beam[F]、w[M*F]、(bf_get_adaptive / bf_get_doa_used / bf_get_bin_resets) ----|
 ─── 重置 / 結束 ────────────────────────────────────────────────────────────────
     |-- srp_reset ; bf_reset ;(VAD:對同一塊 pool 重新 vad_api_init)
     |-- vad_api_destroy ; bf_destroy ; srp_destroy      (BF 在 SRP 之前;pool 路徑的 destroy 不釋放記憶體)

vad_raw                 0      1      1      1      1      1      1      0
vad_out(含 hangover)    0      1      1      1      1      1      1      1

doa_s(smoother)        舊     舊     舊     舊     舊     舊     新     新   ← 新方向連續 3 次 SRP 結果(互差 ≤ angle_tol)才採用

環節延遲依據
VAD 輸出0 hop;median_k 讓起點與終點各延後 floor(k/2) hop;hangover 讓終點再延後 hangover hopVAD/vad.c;hangover=3、median 關/開(k=5) 時，起點 0/+2 hop、終點 +3/+5 hop
SRP / BF 演算0 hop(同 hop 輸入、同 hop 輸出)逐 hop 呼叫,無跨 hop 緩衝
換向從第一次 SRP 觀測到新方向起,需 switch_consec 次連續觀測(互差 ≤ angle_tol),即再經 (switch_consec−1) × update_interval hop;從聲源移動算起最多再加 update_interval−1 hop 等 SRP 輪到(vad_raw 持續為 1 時;上例 3/2 為 4~5 hop);第一次取得 DOA 不需等待doa/doa_smoother.c、doa/srp.c:doa_step

SRP 的大宗是兩張預先計算的表:a_array ≈ A·M·F·8 B 與 pair_steer ≈ A·M(M−1)/2·F·8 B(出處 doa/srp.c:srp_get_mem_size)。因此記憶體隨 num_angles 線性成長,這是 num_angles 不宜隨意調大的主因。BF 的大宗是 P[M][M][F] 的 RLS 協方差,與 num_angles 無關。
